<script>
  (function () {
    var player = document.querySelector(".page-404__player");
    if (!player) return;

    var button = player.querySelector(".page-404__play");
    var status = player.querySelector(".page-404__status");
    var frame = player.querySelector(".page-404__frame");
    var videoId = player.dataset.ytId;

    if (!videoId) {
      player.hidden = true;
      return;
    }

    function stop() {
      frame.textContent = "";
      player.classList.remove("is-playing");
      button.setAttribute("aria-pressed", "false");
      button.setAttribute("aria-label", "Play the track");
      status.textContent = "press play while you're lost";
    }

    function play() {
      var iframe = document.createElement("iframe");
      // The iframe is built here, not in the markup, so nothing reaches YouTube
      // until the click. The click is also what lets autoplay through.
      iframe.src = "https://www.youtube-nocookie.com/embed/" + encodeURIComponent(videoId) + "?autoplay=1&rel=0";
      iframe.allow = "autoplay";
      iframe.title = "KiiiKiii, 404 (New Era)";
      iframe.setAttribute("frameborder", "0");
      frame.appendChild(iframe);
      player.classList.add("is-playing");
      button.setAttribute("aria-pressed", "true");
      button.setAttribute("aria-label", "Pause the track");
      status.textContent = "now playing";
    }

    button.addEventListener("click", function () {
      if (player.classList.contains("is-playing")) stop();
      else play();
    });
  })();
</script>
